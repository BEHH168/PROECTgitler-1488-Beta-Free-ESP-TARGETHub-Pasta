local wallCheckCache = {}
local wallCheckCacheTime = {}
local WALL_CHECK_CACHE_DURATION = 0.15

local wallCheckSettings = {
    MaxRayDistance = 2000,
    VisibilityThreshold = 0.5,
    CheckParts = true,
}

local function isPartVisible(origin, targetPos, targetChar)
    local cacheKey = targetChar
    local currentTime = tick()
    
    if wallCheckCache[cacheKey] and (currentTime - (wallCheckCacheTime[cacheKey] or 0)) < WALL_CHECK_CACHE_DURATION then
        return wallCheckCache[cacheKey]
    end
    
    local il = {}
    if cachedChar then il[#il+1] = cachedChar end
    local cam = Workspace.CurrentCamera
    if cam then il[#il+1] = cam end
    
    for i = 1, #ignoreFolderNames do
        local f = Workspace:FindFirstChild(ignoreFolderNames[i])
        if f then il[#il+1] = f end
    end
    
    il[#il+1] = targetChar
    
    wallRayP.FilterDescendantsInstances = il
    
    local checkPoints = {}
    local humanoid = targetChar:FindFirstChild("Humanoid")
    local hrp = targetChar:FindFirstChild("HumanoidRootPart")
    
    if hrp then
        local basePos = hrp.Position
        
        table.insert(checkPoints, basePos)
        table.insert(checkPoints, basePos + Vector3New(0, 3, 0))
        table.insert(checkPoints, basePos - Vector3New(0, 3, 0))
        table.insert(checkPoints, basePos + Vector3New(2, 0, 0))
        table.insert(checkPoints, basePos - Vector3New(2, 0, 0))
        table.insert(checkPoints, basePos + Vector3New(0, 0, 2))
        table.insert(checkPoints, basePos - Vector3New(0, 0, 2))
        
        if wallCheckSettings.CheckParts then
            for _, partName in ipairs({"Head", "Left Arm", "Right Arm", "Left Leg", "Right Leg"}) do
                local part = targetChar:FindFirstChild(partName)
                if part then
                    table.insert(checkPoints, part.Position)
                end
            end
        end
    else
        table.insert(checkPoints, targetPos)
    end
    
    local visiblePoints = 0
    local totalPoints = #checkPoints
    
    for _, checkPos in ipairs(checkPoints) do
        local dir = checkPos - origin
        local dist = dir.Magnitude
        
        if dist > 0.5 and dist < wallCheckSettings.MaxRayDistance then
            local r = Workspace:Raycast(origin, dir, wallRayP)
            if not r then
                visiblePoints = visiblePoints + 1
            else
                local hitInstance = r.Instance
                
                if hitInstance:IsDescendantOf(targetChar) then
                    visiblePoints = visiblePoints + 1
                else
                    if hitInstance:IsA("BasePart") then
                        if hitInstance.Transparency >= 0.3 then
                            visiblePoints = visiblePoints + 1
                        elseif hitInstance.Transparency >= 0.1 and dist > 500 then
                            visiblePoints = visiblePoints + 1
                        end
                    end
                    
                    local isOtherPlayer = false
                    for plr in next, playerCache do
                        if plr.Character and hitInstance:IsDescendantOf(plr.Character) then
                            isOtherPlayer = true
                            break
                        end
                    end
                    
                    if not isOtherPlayer then
                        if hitInstance:IsA("BasePart") and hitInstance.Size.Magnitude < 3 then
                            visiblePoints = visiblePoints + 0.5
                        end
                    end
                end
            end
        end
    end
    
    local visibilityRatio = visiblePoints / totalPoints
    local result = visibilityRatio >= wallCheckSettings.VisibilityThreshold
    
    if hrp then
        local directDist = (hrp.Position - origin).Magnitude
        
        if directDist < 500 then
            result = visibilityRatio >= 0.6
        else
            result = visibilityRatio >= 0.4
        end
    end
    
    wallCheckCache[cacheKey] = result
    wallCheckCacheTime[cacheKey] = currentTime
    
    return result
end

    wallCheckCacheTime = {}

local function hideEsp(e)
    if not e then return end
    if e.BoxFill then e.BoxFill.Visible = false end
    if e.Box then e.Box.Visible = false end
    if e.BoxOutline then e.BoxOutline.Visible = false end
    if e.Name then e.Name.Visible = false end
    if e.Dist then e.Dist.Visible = false end
    if e.HpBg then e.HpBg.Visible = false end
    if e.Hp then e.Hp.Visible = false end
    if e.Chams then
        pcall(function() e.Chams:Destroy() end)
        e.Chams = nil
    end
end

local function rmEsp(plr)
    local e = espCache[plr]
    if not e then return end
    
    pcall(function()
        if e.BoxFill then e.BoxFill:Remove() end
        if e.Box then e.Box:Remove() end
        if e.BoxOutline then e.BoxOutline:Remove() end
        if e.Name then e.Name:Remove() end
        if e.Dist then e.Dist:Remove() end
        if e.HpBg then e.HpBg:Remove() end
        if e.Hp then e.Hp:Remove() end
        if e.Chams then e.Chams:Destroy() end
    end)
    
    espCache[plr] = nil
end

local function updateESP()
    if not ESP.Options.Enabled then
        for _, e in pairs(espCache) do hideEsp(e) end
        return
    end
    
    if not DrawingOK then return end
    
    local cam = Workspace.CurrentCamera
    if not cam then return end
    
    local camPos = cam.CFrame.Position
    local myRoot = cachedChar and cachedChar:FindFirstChild("HumanoidRootPart")
    local now = tick()
    
    for plr in next, playerCache do
        local ch = plr.Character
        if not ch then
            if espCache[plr] then hideEsp(espCache[plr]) end
            continue
        end
        
        local hrp = ch:FindFirstChild("HumanoidRootPart")
        local hum = ch:FindFirstChild("Humanoid")
        
        if not hrp or not hum or hum.Health <= 0 then
            if espCache[plr] then hideEsp(espCache[plr]) end
            continue
        end
        
        local pos = hrp.Position
        
        if myRoot and (myRoot.Position - pos).Magnitude > ESP.Settings.MaxDistance then
            if espCache[plr] then hideEsp(espCache[plr]) end
            continue
        end
        
        local topS = cam:WorldToViewportPoint(pos + Vector3New(0, 3, 0))
        local botS = cam:WorldToViewportPoint(pos - Vector3New(0, 3, 0))
        
        if topS.Z < 0 and botS.Z < 0 then
            if espCache[plr] then hideEsp(espCache[plr]) end
            continue
        end
        
        if not espCache[plr] then espCache[plr] = mkEsp() end
        local e = espCache[plr]
        if not e then continue end
        
        local h = mathAbs(botS.Y - topS.Y)
        local w = h * 0.6
        local cx = (topS.X + botS.X) * 0.5
        local bx = cx - w * 0.5
        local by = topS.Y
        
        if ESP.Options.Box then
            e.BoxOutline.Visible = true
            e.BoxOutline.Position = Vector2New(bx, by)
            e.BoxOutline.Size = Vector2New(w, h)
            
            if ESP.Options.BoxFill then
                e.BoxFill.Visible = true
                e.BoxFill.Position = Vector2New(bx, by)
                e.BoxFill.Size = Vector2New(w, h)
                e.BoxFill.Transparency = ESP.Settings.BoxFillTransparency
            else
                e.BoxFill.Visible = false
            end
            
            e.Box.Visible = true
            e.Box.Position = Vector2New(bx, by)
            e.Box.Size = Vector2New(w, h)
            
            if ESP.Options.TeamColor then
                local teamColor = getPlayerTeamColor(plr)
                if teamColor then
                    e.Box.Color = teamColor
                    e.BoxFill.Color = teamColor
                else
                    e.Box.Color = ESP.Colors.Box
                    e.BoxFill.Color = ESP.Colors.BoxFill
                end
            else
                e.Box.Color = ESP.Colors.Box
                e.BoxFill.Color = ESP.Colors.BoxFill
            end
        else
            e.Box.Visible = false
            e.BoxOutline.Visible = false
            e.BoxFill.Visible = false
        end
        
        if ESP.Options.Name then
            e.Name.Visible = true
            e.Name.Position = Vector2New(cx, by - 16)
            e.Name.Text = plr.DisplayName
            e.Name.Color = ESP.Colors.Name
            
            if ESP.Options.TeamColor then
                local teamColor = getPlayerTeamColor(plr)
                if teamColor then
                    e.Name.Color = teamColor
                end
            end
        else
            e.Name.Visible = false
        end
        
        if ESP.Options.Distance and myRoot then
            e.Dist.Visible = true
            e.Dist.Position = Vector2New(cx, by + h + 2)
            e.Dist.Text = mathFloor((myRoot.Position - pos).Magnitude) .. "m"
            e.Dist.Color = ESP.Colors.Distance
            
            if ESP.Options.TeamColor then
                local teamColor = getPlayerTeamColor(plr)
                if teamColor then
                    e.Dist.Color = teamColor
                end
            end
        else
            e.Dist.Visible = false
        end
        
        if ESP.Options.HealthBar then
            local barX = bx - 6
            local barW = 3
            local hp = mathClamp(hum.Health / hum.MaxHealth, 0, 1)
            
            e.HpBg.Visible = true
            e.HpBg.Position = Vector2New(barX, by)
            e.HpBg.Size = Vector2New(barW, h)
            
            e.Hp.Visible = true
            e.Hp.Position = Vector2New(barX, by + h * (1 - hp))
            e.Hp.Size = Vector2New(barW, h * hp)
            e.Hp.Color = Color3.fromRGB(255 * (1 - hp), 255 * hp, 0)
        else
            e.HpBg.Visible = false
            e.Hp.Visible = false
        end
        
        if ESP.Options.Chams then
            local isVis = false
            if now - (e.LastVisCheck or 0) >= 0.1 then
                e.LastVisCheck = now
                isVis = isPartVisible(camPos, pos, ch)
                e.VisibleNow = isVis
            else
                isVis = e.VisibleNow
            end
            
            local col = isVis and ESP.Colors.ChamsVisible or ESP.Colors.ChamsHidden
            
            if ESP.Options.TeamColor then
                local teamColor = getPlayerTeamColor(plr)
                if teamColor then
                    col = teamColor
                end
            end
            
            if not e.Chams or not e.Chams.Parent then
                if e.Chams then pcall(function() e.Chams:Destroy() end) end
                local hl = Instance.new("Highlight")
                hl.FillTransparency = 0.5
                hl.OutlineTransparency = 0.8
                hl.FillColor = col
                hl.Adornee = ch
                hl.Parent = ch
                e.Chams = hl
            else
                e.Chams.FillColor = col
            end
        else
            if e.Chams then
                pcall(function() e.Chams:Destroy() end)
                e.Chams = nil
            end
        end
    end
end

function ESP:Start()
    if isRunning then return end
    isRunning = true
    connection = RunService.RenderStepped:Connect(updateESP)
end

function ESP:Stop()
    isRunning = false
    if connection then
        connection:Disconnect()
        connection = nil
    end
    
    for plr in pairs(espCache) do
        rmEsp(plr)
    end
    espCache = {}
end

function ESP:Refresh()
    if isRunning then
        self:Stop()
        self:Start()
    end
end

function ESP:Destroy()
    self:Stop()
    for plr in pairs(playerCache) do
        playerCache[plr] = nil
    end
end

local screenCenter = Vector2New(0, 0)
local function getMousePosition()
    local cam = Workspace.CurrentCamera
    if cam then 
        local vp = cam.ViewportSize
        screenCenter = Vector2New(vp.X * 0.5, vp.Y * 0.5)
    end

local function getPositionOnScreen(pos)
    local cam = Workspace.CurrentCamera
    if not cam then return Vector2New(0, 0), false end
    local p, v = cam:WorldToViewportPoint(pos)
    return Vector2New(p.X, p.Y), v
end

local function distSq2D(a, b)
    local dx, dy = a.X - b.X, a.Y - b.Y
    return dx * dx + dy * dy
end

local function distSq3D(a, b)
    local dx, dy, dz = a.X - b.X, a.Y - b.Y, a.Z - b.Z
    return dx * dx + dy * dy + dz * dz
end

local function getClosestPlayer()
    local cam = Workspace.CurrentCamera
    if not cam then return nil end
    
    local myRoot = cachedChar and cachedChar:FindFirstChild("HumanoidRootPart")
    local myPos = myRoot and myRoot.Position
    local mousePos = getMousePosition()
    local camPos = cam.CFrame.Position
    local best, bestD = nil, nil
    
    for plr in next, playerCache do
        local ch = plr.Character
        if not ch then continue end
        
        local part = ch:FindFirstChild(SilentAim.Settings.TargetPart) or ch:FindFirstChild("HumanoidRootPart") or ch:FindFirstChild("Head")
        if not part or not part.Parent then continue end
        
        local hum = ch:FindFirstChild("Humanoid")
        if not hum or hum.Health <= 0 then continue end
        
        local pp = part.Position
        
        if myPos and distSq3D(myPos, pp) > SilentAim.Settings.MaxDistanceSq then continue end
        
        local sp, vis = getPositionOnScreen(pp)
        if not vis then continue end
        
        local sd = distSq2D(mousePos, sp)
        if SilentAim.Options.UseFOV and sd > SilentAim.Settings.FOVRadiusSq then continue end
        if SilentAim.Options.WallCheck and not isPartVisible(camPos, pp, ch) then continue end
        
        local cd = (SilentAim.Options.Priority == "Distance" and myPos) and distSq3D(myPos, pp) or sd
        if not bestD or cd < bestD then
            best = part
            bestD = cd
        end
    end
    
    return best
end

local lastAim = 0
local silentAimRunning = false
local silentAimConnection = nil

local function silentAimUpdate()
    if LibraryUnloaded then
        if fovCircle then fovCircle.Visible = false end
        if snapline then snapline.Visible = false end
        return
    end
    
    local cam = Workspace.CurrentCamera
    if not cam then return end
    
    local now = tick()
    
    if fovCircle then
        fovCircle.Visible = SilentAim.Options.Enabled and SilentAim.Options.UseFOV
        if fovCircle.Visible then
            local mousePos = getMousePosition()
            fovCircle.Position = mousePos
            fovCircle.Radius = SilentAim.Settings.FOVRadius
            fovCircle.Thickness = SilentAim.Settings.FOVThickness
            
            if SilentAim.Options.FOVMode == "Dynamic" then
                if CurrentAimTarget then
                    fovCircle.Color = SilentAim.Colors.FOVColorOnTarget
                else
                    fovCircle.Color = SilentAim.Colors.FOVColorNoTarget
                end
            else
                fovCircle.Color = SilentAim.Colors.FOVColor
            end
        end
    end
    
    if now - lastAim >= 0.033 then
        lastAim = now
        CurrentAimTarget = SilentAim.Options.Enabled and getClosestPlayer() or nil
    end
    
    if snapline then
        if SilentAim.Options.Enabled and SilentAim.Options.Snapline and CurrentAimTarget then
            local p, vis = cam:WorldToViewportPoint(CurrentAimTarget.Position)
            if vis then
                snapline.Visible = true
                snapline.From = getMousePosition()
                snapline.To = Vector2New(p.X, p.Y)
                snapline.Color = SilentAim.Colors.SnaplineColor
                snapline.Thickness = SilentAim.Settings.SnaplineThickness
            else
                snapline.Visible = false
            end
        else
            snapline.Visible = false
        end
    end
end

function SilentAim:Start()
    if silentAimRunning then return end
    silentAimRunning = true
    silentAimConnection = RunService.RenderStepped:Connect(silentAimUpdate)
end

function SilentAim:Stop()
    silentAimRunning = false
    if silentAimConnection then
        silentAimConnection:Disconnect()
        silentAimConnection = nil
    end
    
    if fovCircle then fovCircle.Visible = false end
    if snapline then snapline.Visible = false end
end

function SilentAim:Refresh()
    if silentAimRunning then
        self:Stop()
        self:Start()
    end
end

function SilentAim:Destroy()
    self:Stop()
    if fovCircle then pcall(function() fovCircle:Remove() end) end
    if snapline then pcall(function() snapline:Remove() end) end
end

local function setupRaycastHook()
    if not hookmetamethod or not getnamecallmethod then return end
    
    local oldNamecall
    oldNamecall = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
        local method = getnamecallmethod()
        if method == "Raycast" and self == Workspace then
            if checkcaller() then return oldNamecall(self, ...) end
            
            local args = {...}
            local origin = args[1]
            local direction = args[2]
            local rayParams = args[3]
            
            if typeof(origin) == "Vector3" and typeof(direction) == "Vector3" then
                local mag = direction.Magnitude
                if mag > 50 then
                    local target = CurrentAimTarget
                    if SilentAim.Options.Enabled and target and target.Parent then
                        local newDir = (target.Position - origin).Unit * mag
                        return oldNamecall(self, origin, newDir, rayParams)
                    end
                end
            end
        end
        return oldNamecall(self, ...)
    end))
end

local lastTH = 0
local function updateTargetHUD(now)
    if not SilentAim.Options.TargetHUD or not SilentAim.Options.Enabled then
        HG.Enabled = false
        return
    end
    
    if now - lastTH < 0.15 then return end
    lastTH = now
    
    local ch = CurrentAimTarget and CurrentAimTarget.Parent
    if ch and ch:FindFirstChild("Humanoid") and ch:FindFirstChild("HumanoidRootPart") and ch.Humanoid.Health > 0 then
        HG.Enabled = true
        local pl = Players:GetPlayerFromCharacter(ch)
        if pl then
            NL.Text = pl.DisplayName
            pcall(function()
                AI.Image = Players:GetUserThumbnailAsync(pl.UserId, Enum.ThumbnailType.HeadShot, Enum.ThumbnailSize.Size150x150)
            end)
        else
            NL.Text = ch.Name
            AI.Image = ""
        end        
        local mr = cachedChar and cachedChar:FindFirstChild("HumanoidRootPart")
        if mr then
            DL.Text = "Distance: " .. string_format("%.1f", (mr.Position - ch.HumanoidRootPart.Position).Magnitude) .. "m"
        end
        
        TweenService:Create(HBF2, TweenInfo.new(0.15), {
            Size = UDim2.new(mathClamp(ch.Humanoid.Health / ch.Humanoid.MaxHealth, 0, 1), 0, 1, 0)
        }):Play()
    else
        HG.Enabled = false
    end
end

    originalSilentAimUpdate()
    updateTargetHUD(tick())
end

local GunMods = {}
GunMods.__index = GunMods

GunMods.Options = {
    NoRecoil = false,
    NoSpread = false,
}

GunMods.OriginalGunRecoil = nil
GunMods.GunRecoilHooked = false
GunMods.SpreadTables = {}
GunMods.OriginalSpreadValues = {}

local function findGunRecoil()
    for a, b in getgc(true) do
        if typeof(b) == "function" and islclosure(b) then
            local c = debug.getinfo(b)
            if c and c.name == "GunRecoil" then
                return b
            end
        end
    end
    return nil
end

function GunMods:ApplyNoRecoil()
    if self.Options.NoRecoil then
        if not self.GunRecoilHooked then
            local recoilFunc = findGunRecoil()
            if recoilFunc then
                self.OriginalGunRecoil = recoilFunc
                hookfunction(recoilFunc, function() end)
                self.GunRecoilHooked = true
            end
        end
    end
end

function GunMods:RemoveNoRecoil()
    if self.GunRecoilHooked and self.OriginalGunRecoil then
        pcall(function()
            hookfunction(self.OriginalGunRecoil, self.OriginalGunRecoil)
        end)
        self.GunRecoilHooked = false
        self.OriginalGunRecoil = nil
    end
end

function GunMods:ApplyNoSpread()
    if self.Options.NoSpread then
        for a, b in getgc(true) do
            if type(b) == "table" then
                pcall(function()
                    setreadonly(b, false)
                end)
                
                if not self.OriginalSpreadValues[b] then
                    self.OriginalSpreadValues[b] = {}
                    
                    if rawget(b, "SpreadMagnitude") ~= nil then
                        self.OriginalSpreadValues[b].SpreadMagnitude = rawget(b, "SpreadMagnitude")
                        rawset(b, "SpreadMagnitude", 0)
                    end
                    if rawget(b, "SpreadRoll") ~= nil then
                        self.OriginalSpreadValues[b].SpreadRoll = rawget(b, "SpreadRoll")
                        rawset(b, "SpreadRoll", 0)
                    end
                    if rawget(b, "ProjectileSpread") ~= nil then
                        self.OriginalSpreadValues[b].ProjectileSpread = rawget(b, "ProjectileSpread")
                        rawset(b, "ProjectileSpread", 0)
                    end
                    
                    local c = rawget(b, "NudgeX")
                    if type(c) == "table" then
                        pcall(function()
                            setreadonly(c, false)
                        end)
                        self.OriginalSpreadValues[b].NudgeX = { Min = c.Min, Max = c.Max }
                        rawset(c, "Min", 0)
                        rawset(c, "Max", 0)
                    end
                    
                    local d = rawget(b, "NudgeY")
                    if type(d) == "table" then
                        pcall(function()
                            setreadonly(d, false)
                        end)
                        self.OriginalSpreadValues[b].NudgeY = { Min = d.Min, Max = d.Max }
                        rawset(d, "Min", 0)
                        rawset(d, "Max", 0)
                    end
                    
                    if rawget(b, "OnesidedXNudge") ~= nil then
                        self.OriginalSpreadValues[b].OnesidedXNudge = rawget(b, "OnesidedXNudge")
                        rawset(b, "OnesidedXNudge", 0)
                    end
                    if rawget(b, "RecoverScale") ~= nil then
                        self.OriginalSpreadValues[b].RecoverScale = rawget(b, "RecoverScale")
                        rawset(b, "RecoverScale", 0)
                    end
                    
                    if next(self.OriginalSpreadValues[b]) then
                        self.SpreadTables[b] = true
                    end
                end
            end
        end
    end
end

function GunMods:RemoveNoSpread()
    for table, savedValues in pairs(self.OriginalSpreadValues) do
        pcall(function()
            setreadonly(table, false)
            
            if savedValues.SpreadMagnitude ~= nil then
                rawset(table, "SpreadMagnitude", savedValues.SpreadMagnitude)
            end
            if savedValues.SpreadRoll ~= nil then
                rawset(table, "SpreadRoll", savedValues.SpreadRoll)
            end
            if savedValues.ProjectileSpread ~= nil then
                rawset(table, "ProjectileSpread", savedValues.ProjectileSpread)
            end
            
            local c = rawget(table, "NudgeX")
            if type(c) == "table" and savedValues.NudgeX then
                pcall(function()
                    setreadonly(c, false)
                end)
                rawset(c, "Min", savedValues.NudgeX.Min)
                rawset(c, "Max", savedValues.NudgeX.Max)
            end
            
            local d = rawget(table, "NudgeY")
            if type(d) == "table" and savedValues.NudgeY then
                pcall(function()
                    setreadonly(d, false)
                end)
                rawset(d, "Min", savedValues.NudgeY.Min)
                rawset(d, "Max", savedValues.NudgeY.Max)
            end
            
            if savedValues.OnesidedXNudge ~= nil then
                rawset(table, "OnesidedXNudge", savedValues.OnesidedXNudge)
            end
            if savedValues.RecoverScale ~= nil then
                rawset(table, "RecoverScale", savedValues.RecoverScale)
            end
        end)
    end
    
    self.OriginalSpreadValues = {}
    self.SpreadTables = {}
end

function GunMods:ToggleNoRecoil(enabled)
    self.Options.NoRecoil = enabled
    if enabled then
        self:ApplyNoRecoil()
    else
        self:RemoveNoRecoil()
    end
end

function GunMods:ToggleNoSpread(enabled)
    self.Options.NoSpread = enabled
    if enabled then
        self:ApplyNoSpread()
    else
        self:RemoveNoSpread()
    end
end
