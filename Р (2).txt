-- Silent Aim: Поиск цели
function t2.value28()
    local bestTarget
    local bestFOV = t2.value10.FOV_RADIUS
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

    for _, player in pairs(Players:GetPlayers()) do
        if player ~= Client and player.Character and not IsWhitelisted(player) then
            if t2.value10.TEAM_CHECK and player.Team == Client.Team then continue end
            
            local targetPart = player.Character:FindFirstChild(t2.value10.HIT_PART) or player.Character:FindFirstChild("Head")
            local humanoid = player.Character:FindFirstChildOfClass("Humanoid")
            
            if targetPart and humanoid and humanoid.Health > 0 then
                local screenPos, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
                if onScreen then
                    local vector = Vector2.new(screenPos.X, screenPos.Y)
                    local dist = (vector - center).Magnitude
                    if dist < bestFOV then
                        bestTarget = targetPart
                        bestFOV = dist
                    end
                end
            end
        end
    end
    return bestTarget
end

-- Silent Aim: Хук
local function SetupSilentAim()
    local castRays = ReplicatedStorage:FindFirstChild("Blaster") and ReplicatedStorage.Blaster:FindFirstChild("Utility") and ReplicatedStorage.Blaster.Utility:FindFirstChild("castRays")
    if castRays and hookfunction then
        local oldFunc
        oldFunc = hookfunction(require(castRays), function(plr, origin, directions, ...)
            if plr == Client and t2.value10.SILENT_AIM and type(directions) == "table" then
                for k, v in pairs(directions) do
                    directions[k] = Vector3.new(0, 0, -1)
                end
            end
            local res = oldFunc(plr, origin, directions, ...)
            
            if plr == Client then
                local target = t2.value28()
                if target then
                    if t2.value10.TRACER then
                        t2.value29(origin, target.Position)
                    end
                    if t2.value10.HIT_CHANCE / 100 > math.random() then
                        if type(res) == "table" then
                            if res.position then
                                res.position = target.Position
                                res.instance = target
                                if target.Parent then
                                    res.taggedHumanoid = target.Parent:FindFirstChildOfClass("Humanoid")
                                end
                                return res
                            elseif res[1] and type(res[1]) == "table" then
                                res[1].position = target.Position
                                res[1].instance = target
                                if target.Parent then
                                    res[1].taggedHumanoid = target.Parent:FindFirstChildOfClass("Humanoid")
                                end
                                return res
                            end
                        end
                    end
                end
                if t2.value10.TRACER and type(res) == "table" then
                    if res.position then
                        t2.value29(origin, res.position)
                    elseif res[1] and res[1].position then
                        t2.value29(origin, res[1].position)
                    end
                end
            end
            return res
        end)
    end
end

-- Tracer
function t2.value29(startPos, endPos)
    if not t2.value10.TRACER then return end
    if (endPos - startPos).Magnitude < 1 then return end
    
    local part = Instance.new("Part")
    part.Size = Vector3.new(0.05, 0.05, 0.05)
    part.Transparency = 1
    part.Anchored = true
    part.CanCollide = false
    part.CanTouch = false
    part.CanQuery = false
    part.Position = startPos
    part.Parent = workspace
    
    local att1 = Instance.new("Attachment", part)
    att1.WorldPosition = startPos
    local att2 = Instance.new("Attachment", part)
    att2.WorldPosition = endPos
    
    local beam = Instance.new("Beam")
    beam.Attachment0 = att1
    beam.Attachment1 = att2
    beam.FaceCamera = true
    beam.LightEmission = 1
    beam.LightInfluence = 0
    
    if t2.value10.TRACER_TYPE == "Red Laser" then
        beam.Texture = "rbxassetid://6060542021"
        beam.TextureMode = Enum.TextureMode.Wrap
        beam.TextureLength = 3
        beam.Width0 = 1.2
        beam.Width1 = 1.2
        beam.Color = ColorSequence.new(Color3.fromRGB(255, 10, 10))
    else
        beam.Texture = "rbxassetid://10259604258"
        beam.TextureMode = Enum.TextureMode.Wrap
        beam.TextureLength = 2
        beam.Width0 = 0.4
        beam.Width1 = 0.4
        beam.Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 255, 255)),
            ColorSequenceKeypoint.new(0.5, Color3.fromRGB(150, 50, 220)),
            ColorSequenceKeypoint.new(1, Color3.fromRGB(255, 255, 255))
        })
    end
    beam.Parent = part
    Debris:AddItem(part, t2.value10.TRACER_LIFE)
end

if Drawing then
    -- Создаем ESP для игроков
    local function CreateESP(player)
        if player == Client or t2.value11.espData[player] then return end
        
        local esp = {
            Box = Drawing.new("Square"),
            Name = Drawing.new("Text"),
            Health = Drawing.new("Square"),
            Dist = Drawing.new("Text")
        }
        
        esp.Box.Filled = false
        esp.Box.Thickness = 1.5
        esp.Box.Visible = false
        
        esp.Name.Size = 12
        esp.Name.Center = true
        esp.Name.Outline = true
        esp.Name.OutlineColor = Color3.fromRGB(0, 0, 0)
        esp.Name.Font = 3
        esp.Name.Visible = false
        
        esp.Health.Filled = true
        esp.Health.Thickness = 1
        esp.Health.Visible = false
        
        esp.Dist.Size = 10
        esp.Dist.Center = true
        esp.Dist.Outline = true
        esp.Dist.OutlineColor = Color3.fromRGB(0, 0, 0)
        esp.Dist.Font = 3
        esp.Dist.Visible = false
        
        t2.value11.espData[player] = esp
    end
    
    for _, player in pairs(Players:GetPlayers()) do
        CreateESP(player)
    end
    
    Players.PlayerAdded:Connect(CreateESP)
    Players.PlayerRemoving:Connect(function(player)
        local esp = t2.value11.espData[player]
        if esp then
            pcall(function()
                esp.Box:Remove()
                esp.Name:Remove()
                esp.Health:Remove()
                esp.Dist:Remove()
            end)
            t2.value11.espData[player] = nil
        end
    end)
    
    -- Обновление ESP
    RunService.RenderStepped:Connect(function()
        if not t2.value10.ESP then
            for _, esp in pairs(t2.value11.espData) do
                esp.Box.Visible = false
                esp.Name.Visible = false
                esp.Health.Visible = false
                esp.Dist.Visible = false
            end
            return
        end
        
        t2.value11.rainbowOffset = (t2.value11.rainbowOffset + 0.005) % 1
        local color = Color3.fromHSV(t2.value11.rainbowOffset, 1, 1)
        
        for player, esp in pairs(t2.value11.espData) do
            local char = player.Character
            if not char or not char.Parent then
                esp.Box.Visible = false
                esp.Name.Visible = false
                esp.Health.Visible = false
                esp.Dist.Visible = false
                continue
            end
            
            local hrp = char:FindFirstChild("HumanoidRootPart")
            local head = char:FindFirstChild("Head")
            if not hrp or not head then
                esp.Box.Visible = false
                esp.Name.Visible = false
                esp.Health.Visible = false
                esp.Dist.Visible = false
                continue
            end
            
            local topPos = head.Position + Vector3.new(0, head.Size.Y / 2, 0)
            local bottomPos = hrp.Position - Vector3.new(0, hrp.Size.Y / 2, 0)
            local rightPos = hrp.Position + hrp.CFrame:VectorToWorldSpace(Vector3.new(hrp.Size.X / 2, 0, 0))
            local leftPos = hrp.Position + hrp.CFrame:VectorToWorldSpace(Vector3.new(-hrp.Size.X / 2, 0, 0))
            
            local topScreen, topVis = Camera:WorldToViewportPoint(topPos)
            local bottomScreen, bottomVis = Camera:WorldToViewportPoint(bottomPos)
            local rightScreen, rightVis = Camera:WorldToViewportPoint(rightPos)
            local leftScreen, leftVis = Camera:WorldToViewportPoint(leftPos)
            
            if not topVis and not bottomVis and not rightVis and not leftVis then
                esp.Box.Visible = false
                esp.Name.Visible = false
                esp.Health.Visible = false
                esp.Dist.Visible = false
                continue
            end
            
            local height = math.abs(topScreen.Y - bottomScreen.Y)
            local width = math.abs(rightScreen.X - leftScreen.X)
            if height < 5 then height = 5 end
            if width < 5 then width = 5 end
            
            local centerX = (leftScreen.X + rightScreen.X) / 2
            local centerY = (topScreen.Y + bottomScreen.Y) / 2
            local dist = (Camera.CFrame.Position - hrp.Position).Magnitude
            
            if t2.value10.ESP_BOX then
                esp.Box.Visible = true
                esp.Box.Size = Vector2.new(width, height)
                esp.Box.Position = Vector2.new(centerX - width / 2, centerY - height / 2)
                esp.Box.Color = color
            else
                esp.Box.Visible = false
            end
            
            if t2.value10.ESP_NAME then
                esp.Name.Visible = true
                esp.Name.Text = player.Name
                esp.Name.Position = Vector2.new(centerX, topScreen.Y - 15)
                esp.Name.Color = color
            else
                esp.Name.Visible = false
            end
            
            if t2.value10.ESP_HEALTH then
                local humanoid = char:FindFirstChildOfClass("Humanoid")
                if humanoid and humanoid.MaxHealth > 0 then
                    local healthPercent = math.clamp(humanoid.Health / humanoid.MaxHealth, 0, 1)
                    esp.Health.Visible = true
                    esp.Health.Size = Vector2.new(2, height * healthPercent)
                    esp.Health.Position = Vector2.new(centerX - width / 2 - 4, centerY + height / 2 - height * healthPercent)
                    esp.Health.Color = Color3.fromRGB(255 * (1 - healthPercent), 255 * healthPercent, 0)
                else
                    esp.Health.Visible = false
                end
            else
                esp.Health.Visible = false
            end
            
            if t2.value10.ESP_DIST then
                esp.Dist.Visible = true
                esp.Dist.Text = math.floor(dist) .. "m"
                esp.Dist.Position = Vector2.new(centerX, bottomScreen.Y + 15)
                esp.Dist.Color = color
            else
                esp.Dist.Visible = false
            end
        end
    end)
end

-- Teleport
function t2.value30()
    if t2.value11.isTeleporting then return end
    local hrp = GetHRP()
    if not hrp then return end
    
    t2.value11.isTeleporting = true
    hrp.CFrame = CFrame.new(hrp.Position + Camera.CFrame.LookVector * t2.value10.TP_DIST)
    task.wait(0.15)
    t2.value11.isTeleporting = false
end

-- Noclip
function t2.value31()
    t2.value11.noclipParts = {}
    for _, part in pairs(workspace:GetDescendants()) do
        if part:IsA("BasePart") and part.CanCollide == true then
            table.insert(t2.value11.noclipParts, part)
            part.CanCollide = false
        end
    end
    t2.value10.NO_CLIP = true
end

function t2.value32()
    for _, part in pairs(t2.value11.noclipParts) do
        pcall(function() part.CanCollide = true end)
    end
    t2.value11.noclipParts = {}
    t2.value10.NO_CLIP = false
end

function t2.value33()
    if t2.value10.NO_CLIP then
        t2.value32()
    else
        t2.value31()
    end
end

-- Furry Drop
function t2.value16()
    local hrp = GetHRP()
    if not hrp then return end
    
    t2.value11.furryDropOriginalCFrame = hrp.CFrame
    hrp.CFrame = hrp.CFrame * CFrame.new(0, -t2.value10.FURRY_DROP_DIST, 0)
    t2.value11.furryDropActive = true
    
    if t2.value11.furryDropConnection then
        t2.value11.furryDropConnection:Disconnect()
    end
    
    t2.value11.furryDropConnection = RunService.Heartbeat:Connect(function()
        local hrp = GetHRP()
        if hrp and t2.value11.furryDropActive then
            hrp.CFrame = t2.value11.furryDropOriginalCFrame * CFrame.new(0, -t2.value10.FURRY_DROP_DIST, 0)
        end
    end)
end

function t2.value17()
    t2.value11.furryDropActive = false
    if t2.value11.furryDropConnection then
        t2.value11.furryDropConnection:Disconnect()
        t2.value11.furryDropConnection = nil
    end
    local hrp = GetHRP()
    if hrp and t2.value11.furryDropOriginalCFrame then
        hrp.CFrame = t2.value11.furryDropOriginalCFrame
        t2.value11.furryDropOriginalCFrame = nil
    end
end

function t2.value18()
    if t2.value11.furryDropActive then
        t2.value17()
    else
        t2.value16()
    end
end

UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    
    if input.KeyCode == Enum.KeyCode[t2.value10.GIGA_KEY] and t2.value10.GIGA_JUMP then
        local hrp = GetHRP()
        if hrp then
            hrp.Velocity = Vector3.new(0, t2.value10.GIGA_POWER, 0)
        end
    end
    
    if input.KeyCode == Enum.KeyCode[t2.value10.TP_KEY] then
        t2.value30()
    end
    
    if input.KeyCode == Enum.KeyCode[t2.value10.FLOAT_KEY] then
        t2.value10.FLOAT = not t2.value10.FLOAT
        if t2.value10.FLOAT then
            if not t2.value11.floatPlatform then
                local platform = Instance.new("Part")
                platform.Size = Vector3.new(5, 0.5, 5)
                platform.Anchored = true
                platform.CanCollide = true
                platform.Transparency = 0.3
                platform.BrickColor = BrickColor.new("Bright blue")
                platform.Parent = workspace
                t2.value11.floatPlatform = platform
            end
        else
            if t2.value11.floatPlatform then
                t2.value11.floatPlatform:Destroy()
                t2.value11.floatPlatform = nil
            end
        end
    end
    
    if input.KeyCode == Enum.KeyCode[t2.value10.NOCLIP_KEY] then
        t2.value33()
    end
    
    if input.KeyCode == Enum.KeyCode[t2.value10.FURRY_DROP_KEY] then
        t2.value18()
    end
end)
